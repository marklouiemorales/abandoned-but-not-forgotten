// Puts a loaded Ingenuity object on Mars at its real coordinates, standing "up" from the surface.
export function placeIngenuityOnMars(object, marsRadius = 1) {
  const { lat, lon } = INGENUITY.location;
  const g = INGENUITY.glb;
  const p = moonSurfacePoint(lat, lon, marsRadius * (1 + g.surfaceOffset), g.lonOffsetDeg);
  object.position.copy(p);
  object.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), p.clone().normalize());
  object.rotateY(THREE.MathUtils.degToRad(g.headingDeg));
  object.scale.setScalar(g.marsScale * marsRadius);
  return object;
}

export default INGENUITY;

// lat/lon (degrees) -> point on a sphere. Assumes longitude 0 faces +Z and +Y is north.
export function moonSurfacePoint(latDeg, lonDeg, radius = 1, lonOffsetDeg = 0) {
  const lat = THREE.MathUtils.degToRad(latDeg);
  const lon = THREE.MathUtils.degToRad(lonDeg + lonOffsetDeg);
  return new THREE.Vector3(
    radius * Math.cos(lat) * Math.sin(lon),
    radius * Math.sin(lat),
    radius * Math.cos(lat) * Math.cos(lon)
  );
}

// Puts a loaded Surveyor 3 object on the Moon at its real coordinates, standing "up" from the surface.
export function placeSurveyor3OnMoon(object, moonRadius = 1) {
  const { lat, lon } = SURVEYOR3.location;
  const g = SURVEYOR3.glb;
  const p = moonSurfacePoint(lat, lon, moonRadius * (1 + g.surfaceOffset), g.lonOffsetDeg);
  object.position.copy(p);
  object.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), p.clone().normalize());
  object.rotateY(THREE.MathUtils.degToRad(g.headingDeg));
  object.scale.setScalar(g.moonScale * moonRadius);
  return object;
}

// src/interaction/planetControls.js
// -----------------------------------------------------------------------------
// Grab and spin ANY planet (Moon or Mars) with mouse or touch, with inertia.
//
// Works with any planet scene that exposes:
//   planet.renderer, planet.camera, planet.ready, planet.state.autoRotate,
//   planet.onUpdate(fn), and a pivot group: planet.pivot (preferred) or
//   planet.moonPivot / planet.marsPivot.
//
// Listeners live on `window` in the CAPTURE phase so dragging works even if an
// invisible overlay sits on top of the canvas. Presses that start on real UI
// (buttons, inputs, signal markers, the equipment panel) are ignored.
//
//   const controls = createPlanetControls(planet, { onDragStart: () => {...} });
//   controls.hold('key')      -> keep auto-rotation off
//   controls.release('key')
//   While hold('selected') is active (a signal is open) dragging is disabled.
//   controls.dispose()
//
// Drag direction feels inverted? pass { invert: true }.
// -----------------------------------------------------------------------------

import * as THREE from 'three';

// The group that holds the planet model (and the signal anchors).
export function getPlanetPivot(planet) {
  return planet.pivot ?? planet.moonPivot ?? planet.marsPivot;
}

// True radius of the planet in pivot space. The scenes normalise the model with a
// bounding sphere, so the real surface may sit below 1 (Moon ~0.58).
export function measurePlanetRadius(pivot, fallback = 0.58) {
  const saved = pivot.rotation.clone();
  pivot.rotation.set(0, 0, 0);
  pivot.updateMatrixWorld(true);
  const box = new THREE.Box3().setFromObject(pivot);
  pivot.rotation.copy(saved);
  pivot.updateMatrixWorld(true);
  if (box.isEmpty()) return fallback;
  const s = box.getSize(new THREE.Vector3());
  return (s.x + s.y + s.z) / 6;
}

const UI_SELECTOR = 'button, a, input, textarea, select, label, [data-no-drag], .signal, .eq-overlay';

export function createPlanetControls(planet, options = {}) {
  const {
    idleResumeMs = 6000,   // 0 = resume right away, Infinity = never auto-rotate again
    maxPitch = 1.4,        // radians the user can tilt toward the poles
    friction = 0.93,       // inertia decay per frame (closer to 1 = longer glide)
    invert = false,
    onDragStart = () => {},
  } = options;

  const canvas = planet.renderer.domElement;
  const pivot = getPlanetPivot(planet);
  if (!pivot) throw new Error('createPlanetControls: planet needs a pivot (planet.pivot / moonPivot / marsPivot)');
  const sign = invert ? -1 : 1;
  const body = document.body;

  const prevTouchAction = document.documentElement.style.touchAction;
  document.documentElement.style.touchAction = 'none';
  canvas.style.touchAction = 'none';
  body.classList.add('planet-draggable');

  let R = 0.58;
  planet.ready.then(() => { R = measurePlanetRadius(pivot); });

  const holds = new Set();
  let idleUntil = 0;
  let dragging = false;
  let started = false;
  let activeId = null;
  let moved = 0;
  let lastX = 0, lastY = 0, lastT = 0;
  let vx = 0, vy = 0;           // radians per frame (pitch, yaw)
  let suppressClick = false;
  let disposed = false;
  let focus = null;             // animated "turn this surface point toward the camera"

  // dir = unit direction in pivot-local space (e.g. an anchor's position, normalised).
  // Assumes the camera looks down -Z (planet front faces +Z), like the scenes do.
  function focusOn(dir, duration = 900) {
    const r = Math.hypot(dir.x, dir.z) || 1e-6;
    const toY = Math.atan2(-dir.x, dir.z);
    const toX = Math.max(-maxPitch, Math.min(maxPitch, Math.atan2(dir.y, r)));
    let dY = toY - pivot.rotation.y;
    dY = Math.atan2(Math.sin(dY), Math.cos(dY));          // shortest way round
    focus = { fromY: pivot.rotation.y, dY, fromX: pivot.rotation.x, dX: toX - pivot.rotation.x, t: 0, duration };
    vx = vy = 0;
  }

  function pxPerRad() {
    const h = canvas.clientHeight || window.innerHeight;
    const tanV = Math.tan(THREE.MathUtils.degToRad(planet.camera.fov / 2));
    const d = planet.camera.position.length();
    return ((h / 2) / tanV) * (R / Math.max(d - R, 0.05));
  }
  const clampPitch = (x) => Math.max(-maxPitch, Math.min(maxPitch, x));

  function onDown(e) {
    if (dragging || holds.has('selected')) return;
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    if (e.target instanceof Element && e.target.closest(UI_SELECTOR)) return;
    focus = null;
    dragging = true;
    started = false;
    activeId = e.pointerId;
    moved = 0;
    vx = vy = 0;
    lastX = e.clientX; lastY = e.clientY; lastT = performance.now();
    try { e.target.setPointerCapture(e.pointerId); } catch (_) { /* not critical */ }
  }

  function onMove(e) {
    if (!dragging || e.pointerId !== activeId) return;
    const dx = e.clientX - lastX;
    const dy = e.clientY - lastY;
    const now = performance.now();
    moved += Math.abs(dx) + Math.abs(dy);
    if (!started && moved > 4) {
      started = true;
      body.classList.add('planet-dragging');
      onDragStart();
    }
    if (!started) return;

    const k = 1 / pxPerRad();
    const dYaw = dx * k * sign;
    const dPitch = dy * k * sign;
    pivot.rotation.y += dYaw;
    pivot.rotation.x = clampPitch(pivot.rotation.x + dPitch);

    const frames = Math.max(now - lastT, 8) / 16.67;
    vy = 0.6 * vy + 0.4 * (dYaw / frames);
    vx = 0.6 * vx + 0.4 * (dPitch / frames);
    lastX = e.clientX; lastY = e.clientY; lastT = now;
  }

  function endDrag(e) {
    if (!dragging || (e && e.pointerId !== activeId)) return;
    dragging = false;
    activeId = null;
    body.classList.remove('planet-dragging');
    if (performance.now() - lastT > 80) vx = vy = 0;   // held still before release: no glide
    idleUntil = performance.now() + idleResumeMs;
    if (started) {                                      // a drag must not count as a click
      suppressClick = true;
      setTimeout(() => { suppressClick = false; }, 0);
    }
  }

  function onClickCapture(e) {
    if (!suppressClick) return;
    e.stopImmediatePropagation();
    e.preventDefault();
    suppressClick = false;
  }

  const onBlur = () => endDrag(null);

  window.addEventListener('pointerdown', onDown, true);
  window.addEventListener('pointermove', onMove, true);
  window.addEventListener('pointerup', endDrag, true);
  window.addEventListener('pointercancel', endDrag, true);
  window.addEventListener('blur', onBlur);
  window.addEventListener('click', onClickCapture, true);

  planet.onUpdate((dt) => {
    if (disposed) return;
    if (focus && !dragging) {
      focus.t += dt * 1000;
      const k = Math.min(1, focus.t / focus.duration);
      const e = 1 - Math.pow(1 - k, 3);                    // ease-out
      pivot.rotation.y = focus.fromY + focus.dY * e;
      pivot.rotation.x = focus.fromX + focus.dX * e;
      if (k >= 1) { focus = null; idleUntil = performance.now() + idleResumeMs; }
      planet.state.autoRotate = false;
      return;
    }
    if (!dragging && (Math.abs(vx) > 1e-5 || Math.abs(vy) > 1e-5)) {
      pivot.rotation.y += vy * dt * 60;
      pivot.rotation.x = clampPitch(pivot.rotation.x + vx * dt * 60);
      const f = Math.pow(friction, dt * 60);
      vx *= f; vy *= f;
    }
    const gliding = Math.abs(vx) + Math.abs(vy) > 1e-4;
    planet.state.autoRotate = !dragging && !gliding && holds.size === 0 && performance.now() >= idleUntil;
  });

  return {
    focusOn,
    hold(key) { holds.add(key); },
    release(key) { holds.delete(key); },
    dispose() {
      disposed = true;
      window.removeEventListener('pointerdown', onDown, true);
      window.removeEventListener('pointermove', onMove, true);
      window.removeEventListener('pointerup', endDrag, true);
      window.removeEventListener('pointercancel', endDrag, true);
      window.removeEventListener('blur', onBlur);
      window.removeEventListener('click', onClickCapture, true);
      body.classList.remove('planet-draggable', 'planet-dragging');
      document.documentElement.style.touchAction = prevTouchAction;
    },
  };
}

// src/interaction/planetSignalMarkers.js
// -----------------------------------------------------------------------------
// Pins clickable HTML markers (with a vector guide line) to lat/lon points ON the
// surface of any planet (Moon or Mars).
//
// - Each signal is raycast onto the real mesh once the model has loaded.
// - Markers are children of the planet pivot, so they rotate with the planet.
// - The vector line is FITTED every frame so it ends inside the planet's disc: it
//   points toward the planet's centre and goes up or down, whichever has more room.
// - Signals on the FAR side show as dashed markers just inside the planet's edge.
//   Clicking one rotates the planet so the signal turns to face you.
// - Unknown signals are white. Once the equipment is found they turn green.
//
//   const markers = createSignalMarkers(planet, document.body, SIGNALS, {
//     controls,
//     onSelect: (signal, markerEl) => { ... },
//   });
//   markers.markFound(id, name)  markers.resume()  markers.dispose()
// -----------------------------------------------------------------------------

import * as THREE from 'three';
import { getPlanetPivot, measurePlanetRadius } from './planetControls.js';

function latLonToDirection(lat, lon) {
  const phi = THREE.MathUtils.degToRad(lat);
  const lam = THREE.MathUtils.degToRad(lon);
  return new THREE.Vector3(
    Math.cos(phi) * Math.sin(lam),
    Math.sin(phi),
    Math.cos(phi) * Math.cos(lam)
  );
}

// vector geometry (px)
const DIAG = 46;      // length of the angled part (45 degrees)
const PAD = 8;        // room above/below the horizontal run for ticks + node
const HEIGHT = DIAG + PAD;
const MIN_RUN = 70;
const MAX_RUN = 260;

function buildVector(run) {
  const x = DIAG + run;
  let ticks = '';
  for (let t = DIAG + 36; t < x - 12; t += 36) ticks += `M${t} ${PAD - 6}v12`;
  return {
    base: `M0 ${HEIGHT} L${DIAG} ${PAD} L${x} ${PAD}`,
    flow: `M${DIAG} ${PAD} L${x} ${PAD}`,
    tick: ticks,
    node: `M${x} ${PAD - 8} L${x + 8} ${PAD} L${x} ${PAD + 8} L${x - 8} ${PAD}Z`,
    width: x + 10,
  };
}

export function createSignalMarkers(planet, container, signals, options = {}) {
  const {
    onSelect = () => {},
    controls = null,
    minFacing = 0.12,    // below this the signal counts as "behind the planet"
    fadeRange = 0.18,    // front markers fade in over this range above minFacing
    edgeMargin = 40,     // px outside the screen before hiding
    behindDepth = 1.0,
  } = options;

  const pivot = getPlanetPivot(planet);
  const hold = (k) => (controls ? controls.hold(k) : (planet.state.autoRotate = false));
  const release = (k) => (controls ? controls.release(k) : (planet.state.autoRotate = true));

  const layer = document.createElement('div');
  layer.className = 'signal-layer';
  container.appendChild(layer);

  let disposed = false;
  let snapped = false;
  let activeId = null;
  let R = 0.58;

  const items = signals.map((sig) => {
    const anchor = new THREE.Object3D();
    pivot.add(anchor);

    const el = document.createElement('div');
    el.className = 'signal signal--right';
    el.setAttribute('role', 'button');
    el.tabIndex = 0;
    el.dataset.id = sig.id;
    el.style.visibility = 'hidden';
    el.innerHTML = `
      <svg class="signal__arrow" viewBox="0 0 40 40" aria-hidden="true">
        <path d="M1 1H11M1 1V11M1 1L40 40"/>
      </svg>
      <span class="signal__dot"></span>
      <span class="signal__tag">
        <strong></strong>
      </span>`;
    el.querySelector('strong').textContent = sig.label;
    layer.appendChild(el);

    const it = {
      sig, anchor, el, side: 'right', up: false, behind: false, found: false,
    };

    const select = (e) => {
      e.stopPropagation();
      if (it.behind) {                                   // far side: turn the planet to it
        if (controls) controls.focusOn(anchor.position.clone().normalize());
        return;
      }
      activeId = sig.id;
      hold('selected');
      items.forEach((o) => o.el.classList.toggle('is-active', o.sig.id === activeId));
      onSelect(sig, el);
    };
    el.addEventListener('click', select);
    el.addEventListener('keydown', (e) => { if (e.key === 'Enter' || e.key === ' ') select(e); });
    el.addEventListener('pointerdown', (e) => e.stopPropagation());   // don't start a planet drag
    el.addEventListener('mouseenter', () => hold('hover'));
    el.addEventListener('mouseleave', () => release('hover'));

    return it;
  });

  // ---- Snap every anchor onto the real surface (once the model exists) ----
  function snapToSurface() {
    R = measurePlanetRadius(pivot);
    pivot.updateMatrixWorld(true);

    const meshes = [];
    pivot.traverse((o) => { if (o.isMesh) meshes.push(o); });

    const centre = pivot.getWorldPosition(new THREE.Vector3());
    const ray = new THREE.Raycaster();

    items.forEach(({ sig, anchor }) => {
      const dir = latLonToDirection(sig.lat, sig.lon);
      const originWorld = pivot.localToWorld(dir.clone().multiplyScalar(R * 3));
      ray.set(originWorld, centre.clone().sub(originWorld).normalize());
      const hit = ray.intersectObjects(meshes, false)[0];
      anchor.position.copy(hit ? pivot.worldToLocal(hit.point.clone()) : dir.multiplyScalar(R * 0.99));
    });
    snapped = true;
  }
  planet.ready.then(() => { if (!disposed) snapToSurface(); });

  function setBehind(it, behind) {
    if (it.behind === behind) return;
    it.behind = behind;
    it.el.classList.toggle('is-behind', behind);
  }

  // ---- Per-frame projection ----
  const world = new THREE.Vector3();
  const normal = new THREE.Vector3();
  const toCam = new THREE.Vector3();
  const ndc = new THREE.Vector3();
  const centreW = new THREE.Vector3();
  const centreNdc = new THREE.Vector3();

  function update() {
    if (disposed || !snapped) return;
    const w = window.innerWidth;
    const h = window.innerHeight;
    const cam = planet.camera;

    // planet centre + silhouette radius on screen
    pivot.getWorldPosition(centreW);
    centreNdc.copy(centreW).project(cam);
    const cx = (centreNdc.x * 0.5 + 0.5) * w;
    const cy = (-centreNdc.y * 0.5 + 0.5) * h;
    const d = Math.max(cam.position.distanceTo(centreW), R * 1.01);
    const tanV = Math.tan(THREE.MathUtils.degToRad(cam.fov / 2));
    const radiusPx = ((h / 2) / tanV) * (R / Math.sqrt(d * d - R * R));

    for (const it of items) {
      const { anchor, el } = it;
      anchor.getWorldPosition(world);
      normal.copy(world).sub(centreW).normalize();
      toCam.copy(cam.position).sub(world).normalize();
      const facing = normal.dot(toCam);

      ndc.copy(world).project(cam);
      let x = (ndc.x * 0.5 + 0.5) * w;
      let y = (-ndc.y * 0.5 + 0.5) * h;

      const behind = facing <= minFacing;
      setBehind(it, behind);

      if (behind) {
        // keep it INSIDE the disc, near the edge, in the direction of the hidden signal
        let dx = x - cx, dy = y - cy;
        const len = Math.hypot(dx, dy);
        if (len < 1) { dx = 0; dy = -1; } else { dx /= len; dy /= len; }
        x = cx + dx * radiusPx * behindDepth;
        y = cy + dy * radiusPx * behindDepth;
      }

      const onScreen = x > -edgeMargin && x < w + edgeMargin && y > -edgeMargin && y < h + edgeMargin;
      if (!onScreen) {
        el.style.visibility = 'hidden';
        el.style.pointerEvents = 'none';
        continue;
      }

      // the tag sits toward the planet's centre, sideways and up/down
      const side = x < cx ? 'right' : 'left';
      const up = y >= cy;           // point below centre -> tag goes up
      if (side !== it.side) {
        el.classList.replace(`signal--${it.side}`, `signal--${side}`);
        it.side = side;
      }
      if (up !== it.up) {
        it.up = up;
        el.classList.toggle('is-up', up);
      }

      el.style.transform = `translate(${x.toFixed(1)}px, ${y.toFixed(1)}px)`;
      el.style.opacity = behind ? '0.9' : String(Math.min(1, (facing - minFacing) / fadeRange));
      el.style.pointerEvents = 'auto';
      el.style.visibility = 'visible';
    }
  }
  planet.onUpdate(update);

  return {
    items,
    markFound(id, name) {
      const it = items.find((i) => i.sig.id === id);
      if (!it) return;
      it.found = true;
      it.el.classList.add('is-found');
      it.el.querySelector('strong').textContent = name || (it.sig.equipment && it.sig.equipment.name) || it.sig.label;
    },
    resume() {
      activeId = null;
      items.forEach((it) => it.el.classList.remove('is-active'));
      release('selected');
    },
    dispose() {
      disposed = true;
      items.forEach(({ anchor }) => anchor.removeFromParent());
      layer.remove();
    },
  };
}

import * as THREE from 'three';

export function setupPlanetRaycaster(planet, onHardwareSelected) {
  const raycaster = new THREE.Raycaster();
  const mouse = new THREE.Vector2();
  const container = planet.renderer.domElement;

  function onClick(event) {
    const rect = container.getBoundingClientRect();
    mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
    mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;

    raycaster.setFromCamera(mouse, planet.camera);
    const intersects = raycaster.intersectObjects(planet.pivot.children, true);

    if (intersects.length > 0) {
      planet.state.autoRotate = false;
      onHardwareSelected(intersects[0].object);
    }
  }

  container.addEventListener('click', onClick);
  return () => container.removeEventListener('click', onClick);
}

// Tiny persistent store for the Archive. It keeps TWO things in localStorage:
//   - explorer  : the user card  { name, id, rank, ... }  (whatever the registration scene gives you)
//   - equipment : one card per discovered equipment, keyed by its id (LUN-01, LUN-02, ...)
//
//   const archive = loadArchive({ name: 'osa', id: 'EXP-001', rank: 'Commander' });
//   saveEquipment(archive, EAGLE);        // adds the card (once) and writes to localStorage
//   Object.keys(archive.equipment).length // how many are saved
//   clearArchive();                       // wipe everything (handy while testing)
// -----------------------------------------------------------------------------

const KEY = 'abf:archive:v1';

export function loadArchive(explorer = {}) {
  let data = null;
  try { data = JSON.parse(localStorage.getItem(KEY) || 'null'); } catch { /* ignore broken / blocked storage */ }
  const archive = { explorer: {}, equipment: {}, ...(data || {}) };
  if (explorer && Object.keys(explorer).length) archive.explorer = { ...archive.explorer, ...explorer };
  if (!archive.explorer.joinedAt) archive.explorer.joinedAt = Date.now();   // shown as the departure date
  persist(archive);
  return archive;
}

export function persist(archive) {
  try { localStorage.setItem(KEY, JSON.stringify(archive)); } catch { /* storage full / disabled: keep in memory */ }
}

// the small "equipment card" that is stored (not the whole entity)
export const snapshot = (e) => ({
  id: e.id,
  planet: e.planet,
  planetCode: e.planetCode,
  name: e.name,
  shortName: e.shortName,
  site: e.location?.site,
  coordinates: e.stats?.coordinates,
  savedAt: Date.now(),
});

export function saveEquipment(archive, entity) {
  if (!entity?.id) return false;
  const isNew = !archive.equipment[entity.id];
  if (isNew) archive.equipment[entity.id] = snapshot(entity);
  persist(archive);
  return isNew;
}

export function clearArchive() {
  try { localStorage.removeItem(KEY); } catch { /* ignore */ }
}

// Barcode drawn from the explorer ID (same ID = same bars), like the boarding pass
function makeBarcode(seed) {
  let n = [...seed].reduce((a, c) => (a * 31 + c.charCodeAt(0)) >>> 0, 7);
  const rnd = () => (n = (n * 1664525 + 1013904223) >>> 0) / 4294967296;
  let x = 0, bars = '';
  while (x < 56) {
    const w = 1 + Math.floor(rnd() * 3.2);
    bars += `<rect x="${x}" y="0" width="${w}" height="100"/>`;
    x += w + 1 + Math.floor(rnd() * 2.2);
  }
  return `<svg viewBox="0 0 56 100" preserveAspectRatio="none" fill="#0b1a3a">${bars}</svg>`;
}

  };

  /* ----- DOM ----- */
  const styleEl = document.createElement('style');
  styleEl.textContent = buildStyles(config.accentColor, config.scrim);
  document.head.appendChild(styleEl);

  const root = el('div', 'dl-root');
  root.innerHTML = `
    <div class="dl-scrim"></div>
    <div class="dl-tools">
      <button class="dl-tool" data-dl="skip" type="button"></button>
      <button class="dl-tool icon" data-dl="mute" type="button" aria-pressed="false">
        <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true">
          <path d="M4 9.5h3.5L12 5.5v13l-4.5-4H4z" fill="currentColor" fill-opacity=".25"/>
          <path class="dl-wave" d="M15.5 9a4 4 0 0 1 0 6M18 6.5a8 8 0 0 1 0 11"/>
          <path class="dl-slash" d="M16 9.5l5 5M21 9.5l-5 5"/>
        </svg>
      </button>
    </div>
    <div class="dl-center" data-dl="center"></div>
    <div class="dl-stack">
      <div class="dl-tasks" data-dl="tasks"></div>
      <div class="dl-box" data-dl="box">
        <div class="dl-portrait" data-dl="portrait">${PORTRAIT_SVG}</div>
        <div class="dl-body">
          <div class="dl-name"><span data-dl="speaker"></span>
            <span class="dl-voice" aria-hidden="true"><i></i><i></i><i></i><i></i></span></div>
          <p class="dl-text" data-dl="text" aria-live="polite"></p>
          <div class="dl-actions" data-dl="actions"></div>
        </div>
      </div>
    </div>
  `;
  container.appendChild(root);

  const q = (k) => root.querySelector(`[data-dl="${k}"]`);
  const center = q('center'), tasksEl = q('tasks'), box = q('box');
  const portrait = q('portrait'), speakerEl = q('speaker'), textEl = q('text'), actions = q('actions');
  const skipBtn = q('skip'), muteBtn = q('mute');

  skipBtn.textContent = t.skip;
  speakerEl.textContent = config.speaker;
  if (config.portraitImage) {
    portrait.innerHTML = '';
    portrait.style.backgroundImage = `url('${config.portraitImage}')`;
  }
  // Paper-style mission list
  const paper = el('div', 'dl-paper');
  const stub = el('div', 'dl-pstub');
  const bars = el('div', 'dl-pbar');
  bars.innerHTML = makeBarcode(String(explorer.id || 'EX'));
  stub.append(bars, el('div', 'dl-pstub-text', `${t.tasksStub} \u00b7 ${explorer.id || ''}`));
  const pmain = el('div', 'dl-pmain');
  const head = el('div', 'dl-phead');
  head.append(el('i', 'dl-plogo'), el('span', null, t.tasksTitle), el('em', null, explorer.id || ''));
  const list = el('div', 'dl-plist');
  config.tasks.forEach((task, i) => {
    const col = el('div', 'dl-ptask');
    const pips = el('div', 'dl-pips');
    for (let p = 0; p < task.total; p++) pips.appendChild(el('i'));
    pips.appendChild(el('em', null, `0/${task.total}`));
    col.append(el('small', null, `Task ${i + 1}`), el('b', null, task.planet), el('span', null, task.text), pips);
    list.appendChild(col);
  });
  pmain.append(head, list);
  paper.append(stub, pmain);
  tasksEl.appendChild(paper);

  /* ----- State ----- */
  let index = -1, token = 0, timers = [], typingId = null, finishTyping = null;
  let audio = null, muted = false;

  const later = (fn, ms) => { const id = setTimeout(fn, ms); timers.push(id); return id; };
  const clearTimers = () => { timers.forEach(clearTimeout); timers = []; };
  const stopVoice = () => { if (audio) { audio.pause(); audio = null; } root.classList.remove('speaking'); };
  const stopTyping = () => { clearInterval(typingId); typingId = null; finishTyping = null; };
  const refreshMute = () => {
    const label = muted ? t.voiceOff : t.voiceOn;
    muteBtn.setAttribute('aria-label', label);
    muteBtn.title = label;
    muteBtn.classList.toggle('is-off', muted);
    muteBtn.setAttribute('aria-pressed', String(muted));
  };
  refreshMute();

  // Plays one mp3. If there is no file / it fails / voice is muted, waits `fallbackMs` instead.
  // `onDone` runs when the voice ends (or the fallback wait is over).
  function speak(src, fallbackMs, onDone) {
    const my = token;
    let finished = false;
    const finish = (delay) => {
      if (finished || my !== token) return;
      finished = true;
      later(() => { if (my === token) onDone?.(); }, delay);
    };
    if (!src || muted) { finish(fallbackMs); return; }
    const a = new Audio(src);
    audio = a;
    const fail = () => { root.classList.remove('speaking'); finish(fallbackMs); };
    a.addEventListener('playing', () => { if (my === token) root.classList.add('speaking'); });
    a.addEventListener('ended', () => { root.classList.remove('speaking'); finish(500); });
    a.addEventListener('error', fail);
    a.play().catch(fail);
  }

  function type(full, done) {
    textEl.textContent = '';
    if (reduceMotion || !config.typeSpeedMs) { textEl.textContent = full; done(); return; }
    let n = 0;
    root.classList.add('typing');
    finishTyping = () => {
      stopTyping();
      textEl.textContent = full;
      root.classList.remove('typing');
      done();
    };
    typingId = setInterval(() => {
      n++;
      textEl.textContent = full.slice(0, n);
      if (n >= full.length) finishTyping();
    }, config.typeSpeedMs);
  }

  const firstSay = () => Math.max(0, steps.findIndex((s) => s.type === 'say'));
  const next = () => go(index + 1);

  /* ----- Steps ----- */
  function go(i) {
    token++;
    clearTimers();
    stopVoice();
    stopTyping();
    const s = steps[i];
    if (!s) return;
    index = i;
    root.className = `dl-root mode-${s.type}${s.showTasks ? ' show-tasks' : ''}`;
    center.className = `dl-center${s.type === 'panel' ? ' is-panel' : ''}`;
    center.replaceChildren();
    actions.replaceChildren();

    if (s.type === 'loading') {
      const status = el('p', 'dl-status', s.messages?.[0] || '');
      const bar = el('div', 'dl-bar');
      const fillBar = el('i');
      bar.appendChild(fillBar);
      center.append(status, bar);
      requestAnimationFrame(() => requestAnimationFrame(() => {
        fillBar.style.transition = `transform ${s.duration}ms linear`;
        fillBar.style.transform = 'scaleX(1)';
      }));
      const msgs = s.messages || [];
      msgs.forEach((m, k) => later(() => { status.textContent = m; }, (s.duration / msgs.length) * k));
      later(next, s.duration + 300);
      return;
    }

    if (s.type === 'panel') {
      center.append(el('h2', 'dl-title', fill(s.title)), el('p', 'dl-sub', fill(s.sub)));
      speak(s.audio, s.hold ?? 3200, next);
      return;
    }

    // say / ask
    type(fill(s.text), () => {
      if (s.type === 'say') {
        const btn = el('button', 'dl-btn ghost', t.next);
        btn.type = 'button';
        btn.addEventListener('click', (e) => { e.stopPropagation(); next(); });
        actions.appendChild(btn);
      } else {
        const decline = el('button', 'dl-btn ghost', s.declineLabel || 'Hear it again');
        const accept = el('button', 'dl-btn primary', s.acceptLabel || 'Accept mission');
        decline.type = accept.type = 'button';
        decline.addEventListener('click', (e) => { e.stopPropagation(); go(firstSay()); });
        accept.addEventListener('click', (e) => { e.stopPropagation(); acceptMission(); });
        actions.append(decline, accept);
        accept.focus({ preventScroll: true });
      }
    });
    speak(s.audio, 0);
  }

  function acceptMission() {
    token++;
    clearTimers();
    stopVoice();
    stopTyping();
    root.className = 'dl-root mode-accepted';
    center.className = 'dl-center';
    const bar = el('div', 'dl-bar');
    const fillBar = el('i');
    bar.appendChild(fillBar);
    center.replaceChildren(el('h2', 'dl-title', t.accepted), el('p', 'dl-sub', t.acceptedSub), bar);
    requestAnimationFrame(() => requestAnimationFrame(() => {
      fillBar.style.transition = 'transform 1.4s linear';
      fillBar.style.transform = 'scaleX(1)';
    }));
    later(() => onAccept({ ...explorer }), 1600);
  }

  /* ----- Input ----- */
  const advance = () => {
    const type_ = steps[index]?.type;
    if (finishTyping) finishTyping();            // first click completes the typing
    else if (type_ === 'say' || type_ === 'panel') next();
  };
  const onBoxClick = (e) => { if (!e.target.closest('button')) advance(); };
  const onCenterClick = () => { if (steps[index]?.type === 'panel') next(); };
  const onKey = (e) => {
    if (e.target.closest?.('button')) return;
    if (['Enter', ' ', 'ArrowRight'].includes(e.key)) { e.preventDefault(); advance(); }
  };
  const onSkip = () => go(firstSay());
  const onMute = () => {
    muted = !muted;
    refreshMute();
    if (muted) {
      stopVoice();
      if (steps[index]?.type === 'panel') go(index); // no audio to wait for, use the timer
    }
  };
  box.addEventListener('click', onBoxClick);
  center.addEventListener('click', onCenterClick);
  window.addEventListener('keydown', onKey);
  skipBtn.addEventListener('click', onSkip);
  muteBtn.addEventListener('click', onMute);

  go(0);

  /* ----- Public API ----- */
  return {
    root,
    goTo: go,
    dispose() {
      token++;
      clearTimers();
      stopVoice();
      stopTyping();
      box.removeEventListener('click', onBoxClick);
      center.removeEventListener('click', onCenterClick);
      window.removeEventListener('keydown', onKey);
      skipBtn.removeEventListener('click', onSkip);
      muteBtn.removeEventListener('click', onMute);
      root.remove();
      styleEl.remove();
    },
  };
}

// Tweak here
const SHIP = {
  length: 1.5,                              // world units
  baseRot: { x: -0.6, y: 0.14, z: 0.1 },    // roll (negative = belly toward camera), yaw, pitch
  y: 0,                                     // height above screen centre (fraction of view height)
  speed: 0.5,                               // overall animation speed
  stars: true,
  flameX: 1.55,                             // x where the flame card starts in the model
  zoomDoneAt: 3.3,                          // seconds after Start when the dialogue begins
};
// Launch after "Mission accepted"
const DEPART_SECONDS = 3.4;                 // length of the smooth fly-away
const DEPART_DISTANCE = 11;                 // how far the ship travels (world units)
const PULLBACK = { x: 0.6, y: 0.15, z: 2.4 }; // how far the camera eases back

// Tries each url in turn. A missing file on the Vite dev server comes back as index.html (HTTP 200),
// so the content is checked too instead of trusting the status code.
async function loadObj(urls) {
  const tried = [];
  for (const url of [].concat(urls)) {
    tried.push(url);
    try {
      const res = await fetch(url);
      const text = res.ok ? await res.text() : '';
      if (text && !text.trimStart().startsWith('<') && /^v\s/m.test(text)) return new OBJLoader().parse(text);
    } catch (e) { /* try the next one */ }
  }
  throw new Error('Shuttle OBJ not found. Tried: ' + tried.join(', '));
}

export function createShuttleStage(container, { objUrls, onLoaded, onError, onZoomed }) {
  const calm = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const amp = calm ? 0.25 : 1;

  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  renderer.setClearColor(0x000000, 1);
  renderer.toneMapping = THREE.ACESFilmicToneMapping;
  renderer.toneMappingExposure = 0.75;
  renderer.shadowMap.enabled = true;
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;
  renderer.domElement.className = 'rm-canvas';
  container.appendChild(renderer.domElement);

  const scene = new THREE.Scene();
  scene.fog = new THREE.Fog(0x000000, 1000, 2000);

  // Space environment for reflections: hard sun panel, blue earthshine below, faint rim
  const envScene = new THREE.Scene();
  envScene.add(new THREE.Mesh(new THREE.SphereGeometry(50, 16, 8), new THREE.MeshBasicMaterial({ color: 0x03050a, side: THREE.BackSide })));
  const panel = (w, h, color, k, pos) => {
    const m = new THREE.Mesh(new THREE.PlaneGeometry(w, h), new THREE.MeshBasicMaterial({ color: new THREE.Color(color).multiplyScalar(k), side: THREE.DoubleSide }));
    m.position.set(...pos); m.lookAt(0, 0, 0); envScene.add(m);
  };
  panel(14, 14, 0xfff4e0, 12, [25, 30, 30]);
  panel(70, 40, 0x2f6bff, 1.6, [0, -40, 10]);
  panel(30, 20, 0x8fb0ff, 0.5, [-40, 10, -20]);
  const pmrem = new THREE.PMREMGenerator(renderer);
  const envTex = pmrem.fromScene(envScene, 0.02).texture;
  pmrem.dispose();
  scene.environment = envTex;

  const camera = new THREE.PerspectiveCamera(35, 1, 0.1, 200);

  // Post: HDR render target (MSAA) + bloom for flame and window glow
  const composer = new EffectComposer(renderer, new THREE.WebGLRenderTarget(innerWidth, innerHeight, { type: THREE.HalfFloatType, samples: 4 }));
  composer.addPass(new RenderPass(scene, camera));
  composer.addPass(new UnrealBloomPass(new THREE.Vector2(innerWidth, innerHeight), 0.45, 0.5, 0.9));
  composer.addPass(new OutputPass());

  // Lighting: hard sun with shadows, blue earthshine from below, warm engine glow
  scene.add(new THREE.AmbientLight(0x141c30, 0.25));
  const key = new THREE.DirectionalLight(0xfff1dc, 1.8); key.position.set(4, 5, 6); scene.add(key);
  key.castShadow = true; key.shadow.mapSize.set(2048, 2048);
  Object.assign(key.shadow.camera, { left: -3.5, right: 3.5, top: 3.5, bottom: -3.5, near: 1, far: 20 });
  key.shadow.bias = -0.0004; key.shadow.normalBias = 0.02;
  const earthshine = new THREE.DirectionalLight(0x3a72ff, 0.7); earthshine.position.set(-2, -6, 3); scene.add(earthshine);
  const backRim = new THREE.DirectionalLight(0x6f94ff, 0.5); backRim.position.set(-6, 2, -5); scene.add(backRim);
  const engineLight = new THREE.PointLight(0xff8a3a, 3.5, 9, 2); scene.add(engineLight);

  // Rig: float -> tilt (orientation) -> pivot (scale/center)
  const rig = new THREE.Group(); scene.add(rig);
  const tilt = new THREE.Group(); rig.add(tilt);
  tilt.rotation.set(SHIP.baseRot.x, SHIP.baseRot.y, SHIP.baseRot.z);

  // Stars
  if (SHIP.stars) {
    const p = new Float32Array(2000 * 3);
    for (let i = 0; i < 2000; i++) {
      const v = new THREE.Vector3().randomDirection().multiplyScalar(90);
      p.set([v.x, v.y, v.z], i * 3);
    }
    const g = new THREE.BufferGeometry(); g.setAttribute('position', new THREE.BufferAttribute(p, 3));
    scene.add(new THREE.Points(g, new THREE.PointsMaterial({ size: 1.5, sizeAttenuation: false, color: 0xffffff, transparent: true, opacity: 0.75, fog: false })));
  }

  // Speed streaks (live in tilt space so they run parallel to the exhaust)
  const N = 26, streaks = [];
  const sPos = new Float32Array(N * 6), sCol = new Float32Array(N * 6);
  for (let i = 0; i < N; i++) {
    const a = Math.random() * Math.PI * 2, r = 2.2 + Math.random() * 6;
    streaks.push({ x: (Math.random() - 0.5) * 30, y: Math.cos(a) * r, z: Math.sin(a) * r, len: 0.6 + Math.random() * 1.2, spd: 5 + Math.random() * 6 });
    sCol.set([0, 0, 0, 0.22, 0.3, 0.5], i * 6);
  }
  const sGeo = new THREE.BufferGeometry();
  sGeo.setAttribute('position', new THREE.BufferAttribute(sPos, 3).setUsage(THREE.DynamicDrawUsage));
  sGeo.setAttribute('color', new THREE.BufferAttribute(sCol, 3));
  const streakMesh = new THREE.LineSegments(sGeo, new THREE.LineBasicMaterial({ vertexColors: true, transparent: true, blending: THREE.AdditiveBlending, depthWrite: false, fog: false }));
  streakMesh.frustumCulled = false; tilt.add(streakMesh);

  // Flame shader (uv.x across the card, uv.y from engine -> tip)
  const flameMat = new THREE.ShaderMaterial({
    uniforms: { uTime: { value: 0 }, uFade: { value: 1 } },
    transparent: true, depthWrite: false, side: THREE.DoubleSide, blending: THREE.AdditiveBlending,
    vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    fragmentShader: `
      uniform float uTime, uFade; varying vec2 vUv;
      float h(vec2 p){ return fract(sin(dot(p, vec2(127.1,311.7))) * 43758.5453); }
      float n(vec2 p){ vec2 i=floor(p), f=fract(p); f=f*f*(3.-2.*f);
        return mix(mix(h(i),h(i+vec2(1,0)),f.x), mix(h(i+vec2(0,1)),h(i+vec2(1,1)),f.x), f.y); }
      float fbm(vec2 p){ float v=0., a=.5; for(int i=0;i<4;i++){ v+=a*n(p); p*=2.03; a*=.5; } return v; }
      void main(){
        float t = vUv.y, x = (vUv.x - 0.5) * 2.0;
        float a1 = fbm(vec2(vUv.x*3.0, t*5.0 - uTime*3.2));
        float a2 = fbm(vec2(vUv.x*6.0 + 7.0, t*9.0 - uTime*5.5));
        float width = mix(0.6, 0.1, pow(t, 0.7));
        float d = abs(x + (a1 - 0.5) * 0.6 * t) / width;
        float body = smoothstep(1.0, 0.0, d);
        float core = smoothstep(0.5, 0.0, d) * (1.0 - t * 0.85);
        float fade = smoothstep(1.0, 0.2, t) * smoothstep(0.0, 0.04, t);
        float al = body * fade * (0.7 + 0.6 * a2) * uFade;
        vec3 c = mix(vec3(1.0,0.97,0.85), vec3(1.0,0.6,0.18), smoothstep(0.0, 0.3, t + (1.0 - core) * 0.25));
        c = mix(c, vec3(1.0,0.27,0.1), smoothstep(0.3, 0.65, t + d * 0.2));
        c = mix(c, vec3(0.3,0.28,0.95), smoothstep(0.62, 1.0, t + d * 0.15));
        c += core * vec3(1.0,0.9,0.7) * 0.7;
        gl_FragColor = vec4(c * 1.0, al);
        #include <tonemapping_fragment>
        #include <colorspace_fragment>
      }`,
  });

  // Material look by name (no .mtl supplied)
  function styleMat(name = '') {
    const k = name.toLowerCase();
    const o = { color: 0x2a2e36, roughness: 0.42, metalness: 0.3, envMapIntensity: 0.7 };
    if (k.includes('whi')) { o.color = 0x0a1114; o.emissive = 0x8fe8ff; o.emissiveIntensity = 1.3; }
    else if (k.includes('winodws_top_blac')) o.color = 0x040506;
    else if (k.includes('bay-win-ins')) { o.emissive = 0x6fdcff; o.emissiveIntensity = 1.4; o.color = 0x06090c; }
    else if (k.includes('bay-win')) o.color = 0x06080a;
    else if (k.includes('glass')) { o.color = 0x04070b; o.metalness = 0.9; o.roughness = 0.12; o.envMapIntensity = 0.9; }
    else if (k.includes('handrail')) { o.color = 0x6a707a; o.metalness = 0.8; o.roughness = 0.35; }
    else if (k.includes('arrows')) { o.emissive = 0xff7a2a; o.emissiveIntensity = 1.5; }
    else if (k.includes('bay')) o.color = 0x2b2f36;
    else if (k.includes('oms') || k.includes('rcs')) o.color = 0x2a2e35;
    else if (k.includes('nose')) o.color = 0x17191d;
    else if (k.includes('belly') || k.includes('wingtop') || k.includes('flap')) o.color = 0x1d2025;
    else if (k.includes('material')) { o.color = 0x23262c; o.metalness = 0.6; }
    return new THREE.MeshPhysicalMaterial({ ...o, clearcoat: k.includes('glass') ? 1 : 0.3, clearcoatRoughness: 0.3 });
  }

  // Load + build
  let flame = null, shuttleObj = null, disposed = false;
  loadObj(objUrls).then((obj) => {
    if (disposed) return;
    try {
      obj.traverse((m) => {
        if (!m.isMesh) return;
        if (m.name === 'Plane') { m.material = flameMat; m.renderOrder = 2; flame = m; return; }
        m.material = Array.isArray(m.material) ? m.material.map((x) => styleMat(x.name)) : styleMat(m.material.name);
        m.castShadow = m.receiveShadow = true;
      });
      const hull = new THREE.Box3();
      obj.traverse((m) => { if (m.isMesh && m !== flame) hull.expandByObject(m); });
      if (hull.isEmpty()) throw new Error('The OBJ file has no meshes');
      const c = hull.getCenter(new THREE.Vector3());
      obj.position.set(-c.x, -c.y, -c.z);
      const pivot = new THREE.Group(); pivot.add(obj);
      pivot.scale.setScalar(SHIP.length / (hull.max.x - hull.min.x));
      tilt.add(pivot);
      shuttleObj = obj;
      // engine light sits just behind the nozzle, in tilt space
      engineLight.userData.local = new THREE.Vector3(SHIP.flameX - c.x - 0.5, 0.84 - c.y, 0).multiplyScalar(pivot.scale.x);
      onLoaded?.();
    } catch (e) {
      console.error(e);
      onError?.(e);
    }
  }).catch((err) => {
    console.error(err);
    onError?.(err);
  });

  // Pointer parallax
  const mouse = new THREE.Vector2(), smooth = new THREE.Vector2();
  const onPointer = (e) => { mouse.set(e.clientX / innerWidth * 2 - 1, -(e.clientY / innerHeight * 2 - 1)); };
  addEventListener('pointermove', onPointer);

  let dist = 9.5, shuttleY = 0;
  function resize() {
    const w = innerWidth, h = innerHeight, aspect = w / h;
    renderer.setSize(w, h, false);
    composer.setSize(w, h);
    camera.aspect = aspect; camera.updateProjectionMatrix();
    dist = 9.5 * Math.max(1, 1.1 / aspect);
    shuttleY = SHIP.y * 2 * dist * Math.tan(THREE.MathUtils.degToRad(camera.fov / 2));
    rig.position.x = 0;
  }
  addEventListener('resize', resize); resize();

  // Timelines
  const COCKPIT = new THREE.Vector3(2.55, 0.92, 0.22);    // zoom target near the nose, in model coordinates
  const ZOOM_OFFSET = new THREE.Vector3(0.3, -0.15, 0.7); // camera end position relative to the cockpit
  const SHIP_K = SHIP.length / 1.8;                       // launch distances were tuned at length 1.8
  const L = { on: false, t: 0, done: false };             // Start button: thrust + zoom
  const D = { on: false, t: 0, cam: null, look: null };   // Mission accepted: depart
  const clamp01 = (x) => Math.min(1, Math.max(0, x));
  const ease = (x) => x * x * (3 - 2 * x);
  const cockpitW = new THREE.Vector3(), basePos = new THREE.Vector3(), endPos = new THREE.Vector3(), look = new THREE.Vector3();
  const clock = new THREE.Clock(), tmp = new THREE.Vector3();

  renderer.setAnimationLoop(() => {
    const dt = Math.min(clock.getDelta(), 0.05), t = clock.elapsedTime * SHIP.speed;
    flameMat.uniforms.uTime.value = t;

    if (L.on) L.t += dt;
    if (D.on) D.t += dt;

    // launch timeline (seconds since Start): thrust 0-1.6, zoom 1.1-3.3, then hold on the cockpit
    const boost = calm ? 0 : ease(clamp01(L.t / 1.6));
    const drift = calm ? 0 : 1.1 * SHIP_K * ease(clamp01(L.t / 2.2));
    const zoom = calm ? 0 : ease(clamp01((L.t - 1.1) / 2.2));
    // depart timeline (seconds since "Mission accepted"): the burn builds gently, the ship
    // accelerates smoothly (no sudden jumps), the camera pulls back and follows a little
    const dep = calm ? 0 : ease(clamp01(D.t / 1.8));                  // 0 -> 1 engine build-up
    const thrust = boost + 1.2 * dep;
    const dU = clamp01(D.t / DEPART_SECONDS);
    const uT = D.t / DEPART_SECONDS;
    const departX = calm ? 0 : DEPART_DISTANCE * uT * uT;            // starts at rest, keeps accelerating
    const calmFx = 1 - 0.85 * dep;                                    // damp wobble/flicker while flying

    if (flame) {
      const s = (1 + (0.05 * Math.sin(t * 23) + 0.04 * Math.sin(t * 37.3) + 0.03 * Math.sin(t * 61)) * calmFx) * (1 + 1.4 * thrust);
      flame.scale.x = s; flame.position.x = SHIP.flameX * (1 - s);
    }
    if (engineLight.userData.local) {
      tmp.copy(engineLight.userData.local); tilt.localToWorld(tmp); engineLight.position.copy(tmp);
    }
    engineLight.intensity = 3.5 * (1 + 2.2 * thrust) + (0.9 * Math.sin(t * 31) + 0.6 * Math.sin(t * 53)) * calmFx;

    rig.position.x = drift + departX;
    rig.position.y = shuttleY + Math.sin(t * 0.9) * 0.1 * amp * calmFx + 0.25 * SHIP_K * boost;
    tilt.rotation.set(
      SHIP.baseRot.x + Math.sin(t * 0.55) * 0.05 * amp * calmFx + Math.sin(t * 40) * 0.0015 * amp * calmFx,
      SHIP.baseRot.y + Math.sin(t * 0.37) * 0.04 * amp * calmFx,
      SHIP.baseRot.z + Math.sin(t * 0.7 + 1) * 0.03 * amp * calmFx + 0.07 * boost);

    for (let i = 0; i < N; i++) {
      const k = streaks[i];
      k.x -= k.spd * dt * SHIP.speed * amp * (1 + 4 * thrust); if (k.x < -16) k.x += 32;
      sPos.set([k.x, k.y, k.z, k.x + k.len, k.y, k.z], i * 6);
    }
    sGeo.attributes.position.needsUpdate = true;

    smooth.lerp(mouse, 1 - Math.pow(0.001, dt));
    basePos.set(smooth.x * 0.8 * amp, 0.15 + smooth.y * 0.45 * amp, dist);

    if (D.on) {
      // smooth: the camera eases back from the cockpit shot while its aim follows the ship a little
      if (!D.cam) { D.cam = camera.position.clone(); D.look = look.clone(); }
      const pull = calm ? 0 : ease(clamp01(D.t / DEPART_SECONDS));
      camera.position.set(
        D.cam.x + PULLBACK.x * pull,
        D.cam.y + PULLBACK.y * pull,
        D.cam.z + PULLBACK.z * pull);
      look.set(D.look.x + departX * 0.5, D.look.y, D.look.z);
      camera.lookAt(look);
    } else if (zoom > 0 && shuttleObj) {
      scene.updateMatrixWorld(true);
      cockpitW.copy(COCKPIT); shuttleObj.localToWorld(cockpitW);
      endPos.copy(cockpitW).addScaledVector(ZOOM_OFFSET, SHIP_K);
      camera.position.lerpVectors(basePos, endPos, zoom);
      look.lerpVectors(tmp.set(0, 0, 0), cockpitW, zoom);
      camera.lookAt(look);
    } else {
      camera.position.copy(basePos);
      camera.lookAt(0, 0, 0);
    }

    if (L.on && !L.done && L.t >= (calm ? 0.3 : SHIP.zoomDoneAt)) { L.done = true; onZoomed?.(); }
    composer.render();
  });

  return {
    launch() { L.on = true; },
    depart() { D.on = true; },
    dispose() {
      disposed = true;
      renderer.setAnimationLoop(null);
      removeEventListener('pointermove', onPointer);
      removeEventListener('resize', resize);
      scene.traverse((o) => {
        o.geometry?.dispose?.();
        const m = o.material;
        (Array.isArray(m) ? m : [m]).forEach((x) => x?.dispose?.());
      });
      envTex.dispose();
      composer.dispose?.();
      renderer.dispose();
      renderer.domElement.remove();
    },
  };
}

// Makes the name/ID screen (#registration-overlay) float on top of the shuttle:
// raised above the 3D canvas, transparent background, plus a soft dark vignette for legibility.
function liftRegistrationLayer() {
  const ov = document.getElementById('registration-overlay');
  const parent = ov?.parentElement;
  const saved = [];
  const set = (node, props) => {
    if (!node) return;
    const prev = {};
    for (const k of Object.keys(props)) { prev[k] = node.style[k]; node.style[k] = props[k]; }
    saved.push([node, prev]);
  };
  if (parent && getComputedStyle(parent).position === 'static') set(parent, { position: 'relative' });
  set(parent, { zIndex: '50' });
  set(ov, { zIndex: '50', background: 'transparent', pointerEvents: 'auto' });

  const scrim = document.createElement('div');
  Object.assign(scrim.style, {
    position: 'fixed', inset: '0', zIndex: '49', pointerEvents: 'none', opacity: '0', transition: 'opacity .8s',
    background: 'radial-gradient(ellipse at center, rgba(2,5,14,.55) 0%, rgba(2,5,14,.25) 55%, rgba(2,5,14,0) 100%)',
  });
  document.body.appendChild(scrim);
  requestAnimationFrame(() => { scrim.style.opacity = '1'; });

  return {
    restore() {
      saved.forEach(([node, prev]) => Object.assign(node.style, prev));
      scrim.remove();
    },
  };
}

/* =============================================================================
   BRIEFING SCENE = intro -> (explorer name) -> dialogue -> launch -> black
   =============================================================================
   createBriefingScene(document.body, {
     stageContainer,                 // where the 3D canvas goes (default: container)
     registerExplorer: (done) => scene,   // optional name screen. Call done(explorer)
                                          // when confirmed; return the scene (it gets .dispose())
     explorer,                       // used when there is no registerExplorer
     onEnd: (explorer) => { ... },   // screen is fully black: start next scene,
                                     // then call revealFromBlack()
     // optional: onAccept, config (dialogue overrides), assets, fadeDelayMs, fadeMs
   });
   Events on `document`:
     'remember-me:launch'  Start journey pressed
     'remember-me:start'   zoom finished (name screen / dialogue begins)
     'remember-me:depart'  mission accepted, ship launches
     'remember-me:end'     screen is fully black
   ============================================================================= */

export function createBriefingScene(container = document.body, options = {}) {
  const reduce = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;
  const assets = {
    // first one that exists wins
    obj: ['/assets/shuttle-fire.obj', '/assets/models/shuttle-fire.obj', '/assets/models/shuttle/shuttle-fire.obj', '/shuttle-fire.obj'],
    logo: '/assets/remember-me-logo.svg',
    ...(options.assets || {}),
  };
  const {
    stageContainer = container,
    registerExplorer,
    config,
    onAccept,
    onEnd,
    fadeDelayMs = reduce ? 200 : 2200,   // accepted -> start of the black fade (ship is leaving)
    fadeMs = reduce ? 300 : 1400,
  } = options;

  let me = {
    name: 'Explorer', id: 'EX-0001', rank: 'Cadet',
    issued: new Date().toISOString().slice(0, 10),
    ...(options.explorer || {}),
  };

  const emit = (name, detail) => document.dispatchEvent(new CustomEvent(name, { detail }));
  const timers = [];
  let dialogue = null, registration = null, liftHandle = null;
  let disposed = false;

  /* ----- DOM ----- */
  const styleEl = document.createElement('style');
  styleEl.textContent = INTRO_STYLES;
  document.head.appendChild(styleEl);

  const loadingEl = document.createElement('div');
  loadingEl.className = 'rm-loading';
  loadingEl.textContent = 'Loading shuttle\u2026';
  container.appendChild(loadingEl);

  const intro = document.createElement('main');
  intro.className = 'rm-intro';
  intro.innerHTML = `
    <div class="rm-stack">
      <h1 class="rm-logo"><img alt="Remember Me"></h1>
      <div class="rm-below"><button class="rm-start" type="button" disabled>Start journey</button></div>
    </div>
    <footer>NASA Space Apps Challenge 2026. By The Trail Blazers.</footer>
  `;
  container.appendChild(intro);

  const logoImg = intro.querySelector('.rm-logo img');
  logoImg.addEventListener('error', () => logoImg.replaceWith(document.createTextNode('Remember Me')));
  logoImg.src = assets.logo;
  const startBtn = intro.querySelector('.rm-start');

  /* ----- after the zoom: name screen (if any), then the dialogue ----- */
  function startDialogue() {
    if (disposed || dialogue) return;
    dialogue = createDialogueScene(container, { explorer: me, config, onAccept: handleAccept });
  }

  function onZoomed() {
    if (disposed || dialogue || registration) return;
    emit('remember-me:start');
    if (!registerExplorer) { startDialogue(); return; }
    const lift = liftHandle = liftRegistrationLayer();
    let finished = false;
    registration = registerExplorer((explorer) => {
      if (finished) return;
      finished = true;
      me = { ...me, ...(explorer || {}) };
      // wait a tick so `registration` is assigned even if done() ran synchronously
      queueMicrotask(() => {
        registration?.dispose?.();
        lift.restore();
        startDialogue();
      });
    }) || {};
  }

  /* ----- 3D stage ----- */
  const stage = createShuttleStage(stageContainer, {
    objUrls: assets.obj,
    onLoaded: () => {
      loadingEl.classList.add('done');
      startBtn.disabled = false;
    },
    onError: (err) => {
      loadingEl.classList.remove('done');
      loadingEl.style.color = '#ff8a8a';
      loadingEl.textContent = err?.message || 'Could not load the shuttle model (check public/assets).';
    },
    onZoomed,
  });

  /* ----- Start journey ----- */
  startBtn.addEventListener('click', () => {
    startBtn.disabled = true;
    intro.classList.add('leaving');
    emit('remember-me:launch');
    stage.launch();
  });

  /* ----- Mission accepted: launch away, fade dialogue, go black ----- */
  function handleAccept(ex) {
    emit('remember-me:depart', ex);
    onAccept?.(ex);
    stage.depart();

    const r = dialogue?.root;
    if (r) {
      r.style.pointerEvents = 'none';
      r.style.transition = 'opacity .8s';
      r.style.opacity = '0';
    }
    timers.push(setTimeout(() => blackoutScreen(fadeMs), fadeDelayMs));
    timers.push(setTimeout(() => { emit('remember-me:end', ex); onEnd?.(ex); }, fadeDelayMs + fadeMs));
  }

  return {
    root: intro,
    // Does NOT remove the black overlay: call revealFromBlack() after the next scene is up.
    dispose() {
      disposed = true;
      timers.forEach(clearTimeout);
      registration?.dispose?.();
      liftHandle?.restore();
      dialogue?.dispose();
      stage.dispose();
      intro.remove();
      loadingEl.remove();
      styleEl.remove();
    },
  };
}

// Starts the dialogue when `startEvent` fires, and when the mission is accepted:
//   1. fires 'remember-me:depart' (your three.js code continues the launch animation)
//   2. fades the dialogue out
//   3. after `fadeDelayMs` fades the screen to black over `fadeMs`
//   4. fires 'remember-me:end'
export function initDialogueFlow(options = {}) {
  const reduce = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;
  const {
    container = document.body,
    startEvent = 'remember-me:start',
    explorer = {},
    config,
    onAccept,
    onEnd,                               // called when the screen is fully black
    fadeDelayMs = reduce ? 200 : 1500,
    fadeMs = reduce ? 300 : 1400,
  } = options;

  const me = {
    name: 'Explorer', id: 'EX-0001', rank: 'Cadet',
    issued: new Date().toISOString().slice(0, 10),
    ...explorer,
  };
  let scene = null;
  const timers = [];
  const emit = (name, detail) => document.dispatchEvent(new CustomEvent(name, { detail }));

  const handleAccept = (ex) => {
    emit('remember-me:depart', ex);
    onAccept?.(ex);
    const r = scene.root;
    r.style.pointerEvents = 'none';
    r.style.transition = 'opacity .8s';
    r.style.opacity = '0';
    timers.push(setTimeout(() => scene?.dispose(), 900));
    timers.push(setTimeout(() => blackoutScreen(fadeMs), fadeDelayMs));
    timers.push(setTimeout(() => { emit('remember-me:end', ex); onEnd?.(ex); }, fadeDelayMs + fadeMs));
  };

  const onStart = () => {
    if (scene) return;                         // only once
    scene = createDialogueScene(container, { explorer: me, config, onAccept: handleAccept });
  };
  document.addEventListener(startEvent, onStart);

  return {
    start: onStart,
    reveal: (ms = 1200) => revealFromBlack(ms),
    dispose() {
      document.removeEventListener(startEvent, onStart);
      timers.forEach(clearTimeout);
      scene?.dispose();
      document.getElementById(BLACK_ID)?.remove();
      scene = null;
    },
  };
}

// -----------------------------------------------------------------------------
// The panel a signal expands into (Moon or Mars). It grows out of the clicked

// with your own UI and may return a cleanup function.

export function createEquipmentPanel(parent, options = {}) {
  const { onClose = () => {}, onOpen = () => {} } = options;

  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const DUR = reduceMotion ? 1 : 560;

  const inner = overlay.querySelector('[data-inner]');
  const slot = overlay.querySelector('[data-slot]');
  const openBtn = overlay.querySelector('[data-open]');
  const closeBtn = overlay.querySelector('[data-close]');

  let closing = false;
  let current = null;

  let cleanup = null;

    if (!originEl || originEl.style.visibility === 'hidden') return null;
    const dot = originEl.querySelector('.signal__dot') || originEl;
    const r = dot.getBoundingClientRect();
    return { x: r.left + r.width / 2, y: r.top + r.height / 2 };

    const r = card.getBoundingClientRect();
    const cx = r.left + r.width / 2;
    const cy = r.top + r.height / 2;
    const o = originPoint() || { x: cx, y: cy + 40 };
    return { dx: o.x - cx, dy: o.y - cy };

  function run(el, frames, opts) {
    const a = el.animate(frames, { fill: 'both', ...opts });
    anims.push(a);
    return a;
  }
  function cancelAnims() { anims.forEach((a) => a.cancel()); anims = []; }

  function fill(signal, found) {
    const eq = signal.equipment || {};
    const index = String(signal.index ?? '').padStart(2, '0');
    text('[data-stub]', 'EQUIPMENT RECORD');
    text('[data-kicker]', `Equipment${signal.index ? ' Â· Signal ' + index : ''}`);
    text('[data-title]', eq.name || 'Unidentified equipment');

    const meta = overlay.querySelector('[data-meta]');
    meta.replaceChildren();
    (eq.meta || []).forEach((m) => {
      const d = document.createElement('div');
      const dt = document.createElement('dt'); dt.textContent = m.label;
      const dd = document.createElement('dd'); dd.textContent = m.value;
      d.append(dt, dd);
      meta.appendChild(d);
    });

    text('[data-status]', found ? 'Status Â· Recovered' : 'Status Â· Not logged');
    openBtn.textContent = found ? 'Open equipment' : 'Inspect equipment';

    slot.replaceChildren();
    if (typeof eq.mount === 'function') {
      const result = eq.mount(slot, eq);
      cleanup = typeof result === 'function' ? result : null;
    } else {
      slot.innerHTML = '<span>Signal locked Â· inspect to analyze</span>';
    }
  }

  function open(signal, el, { found = false } = {}) {
    if (isOpen) return;
    isOpen = true;
    current = signal;
    originEl = el;
    cancelAnims();
    fill(signal, found);

    overlay.hidden = false;
    const { dx, dy } = offsetToOrigin();

    run(overlay, [{ opacity: 0 }, { opacity: 1 }], { duration: DUR * 0.7, easing: 'ease-out' });
    run(card, [
      { transform: `translate(${dx}px, ${dy}px) scale(0.05)`, opacity: 0 },
      { transform: `translate(${dx * 0.25}px, ${dy * 0.25}px) scale(0.7)`, opacity: 1, offset: 0.45 },
      { transform: 'translate(0, 0) scale(1)', opacity: 1 },
    ], { duration: DUR, easing: EASE_OUT });
    run(inner, [{ opacity: 0 }, { opacity: 0, offset: 0.4 }, { opacity: 1 }], { duration: DUR, easing: 'ease-out' });

    openBtn.focus({ preventScroll: true });
  }

  async function close({ silent = false } = {}) {
    if (!isOpen || closing) return;
    closing = true;
    const { dx, dy } = offsetToOrigin();
    cancelAnims();

    const d = DUR * 0.75;
    const finished = [
      run(overlay, [{ opacity: 1 }, { opacity: 0 }], { duration: d, easing: 'ease-in' }),
      run(card, [
        { transform: 'translate(0, 0) scale(1)', opacity: 1 },
        { transform: `translate(${dx}px, ${dy}px) scale(0.05)`, opacity: 0 },
      ], { duration: d, easing: EASE_IN }),
      run(inner, [{ opacity: 1 }, { opacity: 0 }], { duration: d * 0.4, easing: 'ease-in' }),
    ].map((a) => a.finished.catch(() => {}));
    await Promise.all(finished);

    cancelAnims();
    overlay.hidden = true;
    if (cleanup) { cleanup(); cleanup = null; }
    slot.replaceChildren();
    const signal = current;
    current = null;
    originEl = null;
    isOpen = false;
    closing = false;
    if (!silent) onClose(signal);
  }

  closeBtn.addEventListener('click', () => close());
  overlay.addEventListener('click', (e) => { if (e.target === overlay) close(); });

  // Primary action: hand the signal to the caller (which navigates to the equipment page).
  openBtn.addEventListener('click', () => {
    if (!current) return;
    const signal = current;
    onOpen(signal);
  });

  const onKey = (e) => { if (e.key === 'Escape' && isOpen) close(); };
  window.addEventListener('keydown', onKey);

  return {
    slot,
    get isOpen() { return isOpen; },
    open,
    close,
    dispose() {
      window.removeEventListener('keydown', onKey);
      cancelAnims();
      if (cleanup) cleanup();
      overlay.remove();
    },
  };
}

// Paints a small equirectangular surface map (maria + craters). Seeded, so it looks the same every time.
function paintSurfaceMap(canvas, planet) {
  const g = canvas.getContext('2d');
  if (!g) return;
  const W = canvas.width, H = canvas.height, mars = planet === 'mars';
  let s = mars ? 91 : 7;
  const rnd = () => {
    s |= 0; s = (s + 0x6D2B79F5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
  g.fillStyle = mars ? '#a65a3b' : '#9b9b9d';
  g.fillRect(0, 0, W, H);
  for (let i = 0; i < 2600; i++) {                                    // regolith grain
    g.fillStyle = rnd() > 0.5 ? 'rgba(255,255,255,.07)' : 'rgba(0,0,0,.09)';
    g.fillRect(rnd() * W, rnd() * H, 1 + rnd() * 2, 1 + rnd() * 2);
  }
  // dark plains: [lon, lat, radius lon, radius lat] in degrees (Procellarum, Imbrium, Serenitatis, Tranquillitatis, ...)
  const maria = mars
    ? [[-60, 10, 34, 20], [40, -10, 40, 22], [100, 20, 30, 18], [-120, -20, 36, 18]]
    : [[-57, 18, 40, 26], [-17, 33, 28, 16], [18, 28, 14, 11], [23, 8, 18, 12], [51, -3, 12, 9], [59, 17, 9, 7], [-15, -21, 16, 11], [-38, -14, 12, 9]];
  maria.forEach(([lon, lat, rx, ry]) => {
    const x = ((lon + 180) / 360) * W, y = ((90 - lat) / 180) * H;
    const rX = (rx / 360) * W, rY = (ry / 180) * H;
    g.save(); g.translate(x, y); g.scale(1, rY / rX);
    const gr = g.createRadialGradient(0, 0, 0, 0, 0, rX);
    gr.addColorStop(0, mars ? 'rgba(60,25,15,.58)' : 'rgba(40,42,50,.66)');
    gr.addColorStop(0.7, mars ? 'rgba(60,25,15,.36)' : 'rgba(40,42,50,.4)');
    gr.addColorStop(1, 'rgba(0,0,0,0)');
    g.fillStyle = gr; g.beginPath(); g.arc(0, 0, rX, 0, Math.PI * 2); g.fill();
    g.restore();
  });
  for (let i = 0; i < 110; i++) {                                     // craters: dark bowl, lit rim, shadow rim
    const x = rnd() * W, y = rnd() * H, r = 1.5 + Math.pow(rnd(), 3) * 9;
    g.fillStyle = 'rgba(0,0,0,.16)';
    g.beginPath(); g.arc(x, y, r, 0, Math.PI * 2); g.fill();
    g.lineWidth = Math.max(0.8, r * 0.18);
    g.strokeStyle = 'rgba(255,255,255,.32)';
    g.beginPath(); g.arc(x, y, r, Math.PI * 0.75, Math.PI * 1.75); g.stroke();
    g.strokeStyle = 'rgba(0,0,0,.28)';
    g.beginPath(); g.arc(x, y, r, -Math.PI * 0.25, Math.PI * 0.75); g.stroke();
  }
}

  resizeObserver.observe(container);

    scene,
    camera,
    renderer,
    pivot: marsPivot,
    marsPivot,
    ready,
    state,

      renderer.domElement.remove();

// Black space background (lots of stars) + centered, slowly rotating moon.
//
// Requires: three  ->  npm i three
// Model:    put the moon in /public/models/moon.glb (or change modelUrl)
//
// USAGE
//   import { createMoonScene } from './moonScene.js';
//
//   const moon = createMoonScene(document.getElementById('moon-container'), {
//     modelUrl: '/models/moon.glb',   // .glb / .gltf (recommended) or .obj
//   });
//
//   moon.ready.then(() => { /* model is loaded; safe to raycast / add clicks */ });
//
// The container needs a real size, e.g. width: 100vw; height: 100vh; position: relative.
//
// FOR THE CLICK FEATURE
//   moon.scene / moon.camera / moon.renderer  -> standard three.js objects
//   moon.moonPivot                            -> Group at the origin that spins; raycast against its children
//   moon.state.autoRotate = false             -> pause spinning (e.g. while navigating)
//   moon.onUpdate((dt) => { ... })            -> run your own code every frame
//   moon.dispose()                            -> cleanup (call on unmount)

import * as THREE from 'three';
import { GLTFLoader } from 'three/addons/loaders/GLTFLoader.js';
import { OBJLoader } from 'three/addons/loaders/OBJLoader.js';

export function createMoonScene(container, options = {}) {
  const {
    modelUrl = '/models/moon.glb',
    textureUrl = null,        // only used for .obj
    fill = 2.0,               // moon size vs screen height (raise to zoom in more)
    rotationSpeed = 0.0008,   // radians per frame
    starsDim = 9000,          // tiny faint stars
    starsMid = 2500,          // medium stars
    starsBright = 300,        // few bright stars
    forceMatte = true,        // moon has no shine
    fov = 45,
  } = options;

  const state = { autoRotate: true, rotationSpeed };

  // ---------- Core ----------
  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(fov, 1, 0.1, 300);
  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.setClearColor(0x000000); // pure black
  renderer.domElement.style.display = 'block';
  container.appendChild(renderer.domElement);

  // ---------- Stars ----------
  const spriteCanvas = document.createElement('canvas');
  spriteCanvas.width = spriteCanvas.height = 64;
  const sg = spriteCanvas.getContext('2d');
  const sgrad = sg.createRadialGradient(32, 32, 0, 32, 32, 32);
  sgrad.addColorStop(0, 'rgba(255,255,255,1)');
  sgrad.addColorStop(0.35, 'rgba(255,255,255,0.45)');
  sgrad.addColorStop(1, 'rgba(255,255,255,0)');
  sg.fillStyle = sgrad;
  sg.fillRect(0, 0, 64, 64);
  const starSprite = new THREE.CanvasTexture(spriteCanvas);

  const PALETTE = [
    [0.72, 0.82, 1.0],  // blue-white
    [1.0, 1.0, 1.0],    // white
    [1.0, 0.9, 0.76],   // warm
    [1.0, 0.74, 0.58],  // orange (rare)
  ];

  const starGeos = [];
  function starLayer({ count, size, minB, power }) {
    const R = 120;
    const pos = new Float32Array(count * 3);
    const col = new Float32Array(count * 3);

    for (let i = 0; i < count; i++) {
      const u = Math.random() * 2 - 1;
      const t = Math.random() * Math.PI * 2;
      const s = Math.sqrt(1 - u * u);
      pos.set([R * s * Math.cos(t), R * u, R * s * Math.sin(t)], i * 3);

      const b = minB + Math.pow(Math.random(), power) * (1 - minB);
      const r = Math.random();
      const p = r < 0.3 ? PALETTE[0] : r < 0.75 ? PALETTE[1] : r < 0.95 ? PALETTE[2] : PALETTE[3];
      col.set([p[0] * b, p[1] * b, p[2] * b], i * 3);
    }

    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
    const pts = new THREE.Points(
      geo,
      new THREE.PointsMaterial({
        size,
        sizeAttenuation: false,
        map: starSprite,
        vertexColors: true,
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
        toneMapped: false,
      })
    );
    pts.frustumCulled = false;
    starGeos.push({ geo, base: col.slice(), pos });
    return pts;
  }

  const stars = new THREE.Group();
  stars.add(
    starLayer({ count: starsDim, size: 1.5, minB: 0.2, power: 2 }),
    starLayer({ count: starsMid, size: 2.4, minB: 0.35, power: 2 }),
    starLayer({ count: starsBright, size: 3.8, minB: 0.7, power: 1.2 })
  );
  scene.add(stars);

  // ---------- Lights: hard sun + faint fill ----------
  const sun = new THREE.DirectionalLight(0xfff6ea, 3.2);
  sun.position.set(4, 2, 6);
  scene.add(sun);
  scene.add(new THREE.AmbientLight(0x1a2233, 0.8));

  // ---------- Moon ----------
  // moonPivot spins at the origin. Its child is the model, scaled so radius = 1.
  const moonPivot = new THREE.Group();
  scene.add(moonPivot);
  const moonMat = new THREE.MeshStandardMaterial({ color: 0xbfbfbf, roughness: 1, metalness: 0 });

  function addModel(obj) {
    const box = new THREE.Box3().setFromObject(obj);
    const sphere = box.getBoundingSphere(new THREE.Sphere());
    obj.position.sub(sphere.center);
    const holder = new THREE.Group();
    holder.add(obj);
    holder.scale.setScalar(1 / sphere.radius);
    moonPivot.add(holder);
    return holder;
  }

  function placeholderMoon() {
    console.warn(`[moonScene] Could not load ${modelUrl} - showing placeholder moon.`);
    const geo = new THREE.IcosahedronGeometry(1, 60);
    const p = geo.attributes.position;
    const v = new THREE.Vector3();
    for (let i = 0; i < p.count; i++) {
      v.fromBufferAttribute(p, i).normalize();
      const n = Math.sin(v.x * 9) * Math.sin(v.y * 9) * Math.sin(v.z * 9);
      v.multiplyScalar(1 + n * 0.015);
      p.setXYZ(i, v.x, v.y, v.z);
    }
    geo.computeVertexNormals();
    return addModel(new THREE.Mesh(geo, moonMat));
  }

  const ext = modelUrl.split('?')[0].split('.').pop().toLowerCase();

  const ready = new Promise((resolve) => {
    if (ext === 'glb' || ext === 'gltf') {
      new GLTFLoader().load(
        modelUrl,
        (gltf) => {
          gltf.scene.traverse((c) => {
            if (!c.isMesh) return;
            if (!c.geometry.attributes.normal) c.geometry.computeVertexNormals();
            if (forceMatte && c.material) {
              c.material.roughness = 1;
              c.material.metalness = 0;
            }
          });
          resolve(addModel(gltf.scene));
        },
        undefined,
        () => resolve(placeholderMoon())
      );
    } else {
      if (textureUrl) {
        new THREE.TextureLoader().load(textureUrl, (tex) => {
          tex.colorSpace = THREE.SRGBColorSpace;
          moonMat.map = tex;
          moonMat.color.set(0xffffff);
          moonMat.needsUpdate = true;
        });
      }
      new OBJLoader().load(
        modelUrl,
        (obj) => {
          obj.traverse((c) => {
            if (!c.isMesh) return;
            if (!c.geometry.attributes.normal) c.geometry.computeVertexNormals();
            c.material = moonMat;
          });
          resolve(addModel(obj));
        },
        undefined,
        () => resolve(placeholderMoon())
      );
    }
  });

  // ---------- Fixed camera: moon centered, sized by `fill` ----------
  function fitCamera() {
    const w = container.clientWidth || window.innerWidth;
    const h = container.clientHeight || window.innerHeight;
    renderer.setSize(w, h);
    camera.aspect = w / h;
    camera.updateProjectionMatrix();

    const tanV = Math.tan(THREE.MathUtils.degToRad(fov / 2));
    const tanFit = Math.min(tanV, tanV * camera.aspect) * fill;
    const dist = 1 / Math.sin(Math.atan(tanFit)); // moon radius is 1
    camera.position.set(0, 0, dist);
    camera.lookAt(0, 0, 0);
  }
  fitCamera();

  const resizeObserver = new ResizeObserver(fitCamera);
  resizeObserver.observe(container);

  let tween = null;
  const starDimDir = new THREE.Vector3();
  const starWorldInv = new THREE.Quaternion();

  function applyStarDim(dir, amount) {
    for (const { geo, base, pos } of starGeos) {
      const col = geo.attributes.color.array;
      for (let i = 0; i < pos.length; i += 3) {
        const d = (pos[i] * dir.x + pos[i + 1] * dir.y + pos[i + 2] * dir.z) / 120;
        const f = 1 - amount * THREE.MathUtils.smoothstep(d, 0.8, 1.0);
        col[i] = base[i] * f;
        col[i + 1] = base[i + 1] * f;
        col[i + 2] = base[i + 2] * f;
      }
      geo.attributes.color.needsUpdate = true;
    }
  }

  // screenX / screenY: 0..1 across and down the screen (0.75, 0.25 = upper right)
  // size: final diameter as a fraction of screen height (0.3 = 30%)
  // depth: how far the Moon recedes, as a multiple of its starting distance
  function recedeTo({
    screenX = 0.75, screenY = 0.25, size = 0.3, depth = 1.7, dim = 0.3, duration = 1400,
  } = {}) {
    const camZ = camera.position.z;
    const endDist = camZ * depth;
    const halfH = endDist * Math.tan(THREE.MathUtils.degToRad(camera.fov / 2));
    const halfW = halfH * camera.aspect;
    const nx = screenX * 2 - 1;
    const ny = 1 - screenY * 2;
    const toPos = new THREE.Vector3(nx * halfW, ny * halfH, camZ - endDist);
    const toScale = size * halfH;

    // star directions are stored in the stars group's local space, so convert the target there
    starDimDir.copy(toPos).normalize().applyQuaternion(starWorldInv.copy(stars.quaternion).invert());

    return new Promise((resolve) => {
      tween = {
        t: 0,
        duration,
        dim,
        fromPos: moonPivot.position.clone(),
        toPos,
        fromScale: moonPivot.scale.x,
        toScale,
        dimDir: starDimDir.clone(),
        resolve,
      };
    });
  }

  function stepTween(dt) {
    if (!tween) return;
    tween.t += dt * 1000;
    const k = Math.min(1, tween.t / tween.duration);
    const e = k < 0.5 ? 4 * k * k * k : 1 - Math.pow(-2 * k + 2, 3) / 2; // ease-in-out cubic
    moonPivot.position.lerpVectors(tween.fromPos, tween.toPos, e);
    moonPivot.scale.setScalar(THREE.MathUtils.lerp(tween.fromScale, tween.toScale, e));
    applyStarDim(tween.dimDir, tween.dim * e);
    if (k >= 1) {
      const done = tween.resolve;
      tween = null;
      done();
    }
  }

  const updaters = [];
  updaters.push(stepTween);
  const clock = new THREE.Clock();

  renderer.setAnimationLoop(() => {
    const dt = clock.getDelta();
    if (state.autoRotate) moonPivot.rotation.y += state.rotationSpeed;
    stars.rotation.y += 0.00003; // very slow star drift
    updaters.forEach((fn) => fn(dt));
    renderer.render(scene, camera);
  });

  // ---------- Public API ----------
  return {
    scene,
    camera,
    renderer,
    pivot: moonPivot,
    moonPivot,
    ready,
    state,
    recedeTo,
    onUpdate(fn) {
      updaters.push(fn);
    },
    dispose() {
      renderer.setAnimationLoop(null);
      resizeObserver.disconnect();
      scene.traverse((o) => {
        if (o.geometry) o.geometry.dispose();
        if (o.material) {
          (Array.isArray(o.material) ? o.material : [o.material]).forEach((m) => {
            if (m.map) m.map.dispose();
            m.dispose();
          });
        }
      });
      renderer.dispose();
      renderer.domElement.remove();
    },
  };
}

// Barcode drawn from the explorer ID, so the same ID always gives the same bars
function makeBarcode(seed) {
  let s = [...seed].reduce((a, c) => (a * 31 + c.charCodeAt(0)) >>> 0, 7);
  const rnd = () => (s = (s * 1664525 + 1013904223) >>> 0) / 4294967296;
  let x = 0;
  let bars = '';
  while (x < 56) {
    const w = 1 + Math.floor(rnd() * 3.2);
    const gap = 1 + Math.floor(rnd() * 2.2);
    bars += `<rect x="${x}" y="0" width="${w}" height="100"/>`;
    x += w + gap;
  }
  return `<svg viewBox="0 0 56 100" preserveAspectRatio="none" fill="#0b1a3a">${bars}</svg>`;
}

      pos[i * 3 + 1] = r * Math.sin(phi) * Math.sin(theta);
      pos[i * 3 + 2] = r * Math.cos(phi) - 10;

      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,

  button.addEventListener('click', submit);

    const h = root.clientHeight || window.innerHeight;
    renderer.setSize(w, h, false); // CSS already sizes the canvas

  });

      input.removeEventListener('keydown', onKey);

        if (name) li.querySelector('span').textContent = name;
      }
      const seg = [...root.querySelectorAll('[data-seg]')].find((n) => n.dataset.seg === id);
      if (seg) seg.classList.add('is-found');
      root.querySelectorAll('[data-obj-found]').forEach((n) => { n.textContent = found.size; });
      root.querySelector('[data-total-found]').textContent = overallFound;
    },
    dim(on) {
      root.classList.toggle('is-dim', !!on);
    },
    hideHint() {},   // hint removed
    dispose() {
      root.remove();
    },
  };
}

// src/scenes/spaceBackground.js
// -----------------------------------------------------------------------------
// The same space background as the registration ID card scene: dark nebula +
// twinkling stars + soft mouse parallax. Own renderer, own loop, own dispose().
//
//   const bg = createSpaceBackground(someDiv);                  // animated (registration card)
//   const bg = createSpaceBackground(someDiv, { still: true }); // FROZEN: no mouse parallax, no drift, no twinkle
//   bg.dispose();
// -----------------------------------------------------------------------------

import * as THREE from 'three';

const NEBULA_VERT = /* glsl */ `
  varying vec2 vUv;
  void main(){ vUv = uv; gl_Position = vec4(position.xy, 0.9999, 1.0); }
`;
const NEBULA_FRAG = /* glsl */ `
  precision highp float;
  varying vec2 vUv;
  uniform float uTime;
  uniform vec3 uColorA;
  uniform vec3 uColorB;
  float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1,311.7))) * 43758.5453); }
  float noise(vec2 p){
    vec2 i = floor(p), f = fract(p);
    vec2 u = f*f*(3.0-2.0*f);
    return mix(mix(hash(i), hash(i+vec2(1,0)), u.x),
               mix(hash(i+vec2(0,1)), hash(i+vec2(1,1)), u.x), u.y);
  }
  float fbm(vec2 p){
    float v = 0.0, a = 0.5;
    for(int i=0;i<5;i++){ v += a*noise(p); p *= 2.0; a *= 0.5; }
    return v;
  }
  void main(){
    vec2 p = vUv * 2.0 - 1.0;
    float n  = fbm(p * 1.4 + vec2(uTime * 0.008, -uTime * 0.006));
    float n2 = fbm(p * 2.8 - vec2(uTime * 0.012, 0.0));
    vec3 col = mix(uColorA, uColorB, n) * pow(n2, 2.4) * 0.8;
    float vignette = smoothstep(1.6, 0.25, length(p));
    gl_FragColor = vec4(col * vignette, 1.0);
  }
`;
const STAR_VERT = /* glsl */ `
  attribute float aSize;
  attribute float aPhase;
  uniform float uTime;
  uniform float uPixelRatio;
  varying float vTwinkle;
  void main(){
    vTwinkle = 0.55 + 0.45 * sin(uTime * 1.4 + aPhase);
    gl_PointSize = aSize * uPixelRatio * vTwinkle;
    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
  }
`;
const STAR_FRAG = /* glsl */ `
  varying float vTwinkle;
  void main(){
    float d = length(gl_PointCoord - 0.5);
    float a = smoothstep(0.5, 0.0, d) * vTwinkle;
    gl_FragColor = vec4(vec3(0.85, 0.92, 1.0), a);
  }
`;

export function createSpaceBackground(container, options = {}) {
  // still: true  -> the background never moves (no mouse parallax, no star rotation, no twinkle, no nebula drift).
  //                 It is drawn once and again only when the window is resized.
  const { starCountFar = 1600, starCountNear = 350, parallaxStrength = 0.4, still = false } = options;

  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(55, 1, 0.1, 200);
  camera.position.set(0, 0, 6);
  camera.lookAt(0, 0, -4);

  const renderer = new THREE.WebGLRenderer({ antialias: true });
  const pixelRatio = Math.min(window.devicePixelRatio, 2);
  renderer.setPixelRatio(pixelRatio);
  renderer.setClearColor(0x000000);
  const canvas = renderer.domElement;
  canvas.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;display:block;';
  container.appendChild(canvas);

  const disposables = [];
  const track = (o) => { disposables.push(o); return o; };

  const nebulaMat = track(new THREE.ShaderMaterial({
    vertexShader: NEBULA_VERT, fragmentShader: NEBULA_FRAG,
    depthTest: false, depthWrite: false,
    uniforms: {
      uTime: { value: 0 },
      uColorA: { value: new THREE.Color('#0a1a45') },
      uColorB: { value: new THREE.Color('#34195a') },
    },
  }));
  const nebula = new THREE.Mesh(track(new THREE.PlaneGeometry(2, 2)), nebulaMat);
  nebula.frustumCulled = false;
  nebula.renderOrder = -10;
  scene.add(nebula);

  function makeStars(count, rMin, rMax, sMin, sMax) {
    const pos = new Float32Array(count * 3);
    const size = new Float32Array(count);
    const phase = new Float32Array(count);
    for (let i = 0; i < count; i++) {
      const r = rMin + Math.random() * (rMax - rMin);
      const theta = Math.random() * Math.PI * 2;
      const phi = Math.acos(2 * Math.random() - 1);
      pos[i * 3] = r * Math.sin(phi) * Math.cos(theta);
      pos[i * 3 + 1] = r * Math.sin(phi) * Math.sin(theta);
      pos[i * 3 + 2] = r * Math.cos(phi) - 10;
      size[i] = sMin + Math.random() * (sMax - sMin);
      phase[i] = Math.random() * Math.PI * 2;
    }
    const geo = track(new THREE.BufferGeometry());
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    geo.setAttribute('aSize', new THREE.BufferAttribute(size, 1));
    geo.setAttribute('aPhase', new THREE.BufferAttribute(phase, 1));
    const mat = track(new THREE.ShaderMaterial({
      vertexShader: STAR_VERT, fragmentShader: STAR_FRAG,
      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
      uniforms: { uTime: { value: 0 }, uPixelRatio: { value: pixelRatio } },
    }));
    const points = new THREE.Points(geo, mat);
    scene.add(points);
    return points;
  }
  const farStars = makeStars(starCountFar, 40, 100, 1, 3);
  const nearStars = makeStars(starCountNear, 12, 30, 1.5, 4.5);

  const pointer = new THREE.Vector2();
  const targetPointer = new THREE.Vector2();
  const onPointerMove = (e) => {
    targetPointer.set((e.clientX / window.innerWidth) * 2 - 1, -((e.clientY / window.innerHeight) * 2 - 1));
  };
  if (!still) window.addEventListener('pointermove', onPointerMove);

  function fit() {
    const w = container.clientWidth || window.innerWidth;
    const h = container.clientHeight || window.innerHeight;
    renderer.setSize(w, h, false);
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
    if (still) renderer.render(scene, camera);   // frozen: redraw only when the size changes
  }
  const resizeObserver = new ResizeObserver(fit);
  resizeObserver.observe(container);
  fit();

  if (!still) {
    const clock = new THREE.Clock();
    renderer.setAnimationLoop(() => {
      const dt = clock.getDelta();
      const time = clock.elapsedTime;
      nebulaMat.uniforms.uTime.value = time;
      farStars.material.uniforms.uTime.value = time;
      nearStars.material.uniforms.uTime.value = time;
      pointer.lerp(targetPointer, 1 - Math.pow(0.001, dt));
      camera.position.set(pointer.x * parallaxStrength, pointer.y * parallaxStrength * 0.6, 6);
      camera.lookAt(0, 0, -4);
      farStars.rotation.y = time * 0.004;
      nearStars.rotation.y = time * 0.009;
      renderer.render(scene, camera);
    });
  }

  return {
    dispose() {
      renderer.setAnimationLoop(null);
      resizeObserver.disconnect();
      window.removeEventListener('pointermove', onPointerMove);
      disposables.forEach((d) => d.dispose?.());
      renderer.dispose();
      canvas.remove();
    },
  };
}

export const GAME_PHASES = Object.freeze([
  'registration',
  'briefing',
  'spaceship',
  'launch',
  'moon',
  'transit',
  'marsLanding',
  'equipment',
  'mars',
  'return',
  'certificate',
]);

export function createGameState(phases) {
  const state = {
    explorer: null,
    phase: null,
    discovered: new Set(),
  };

  let activePhase = null;
  let activeScene = null;

  function exitActivePhase() {
    const previousPhase = activePhase;
    const previousScene = activeScene;
    activePhase = null;
    activeScene = null;
    previousPhase?.exit?.({ state, scene: previousScene });
  }

  function goTo(phase, context = {}) {
    const nextPhase = phases[phase];
    if (!GAME_PHASES.includes(phase) || !nextPhase) {
      throw new Error(`Game phase "${phase}" is not available.`);
    }

    exitActivePhase();
    if (Object.prototype.hasOwnProperty.call(context, 'explorer')) {
      state.explorer = context.explorer;
    }

    state.phase = phase;
    try {
      activeScene = nextPhase.enter({ ...context, state, goTo });
      activePhase = nextPhase;
    } catch (error) {
      state.phase = null;
      throw error;
    }
  }

  function dispose() {
    exitActivePhase();
    state.phase = null;
  }

  return { state, goTo, dispose };
}
